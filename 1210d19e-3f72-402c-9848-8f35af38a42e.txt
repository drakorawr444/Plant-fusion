/* Garden Fusion is an original browser tower-defense prototype.
   It uses Canvas shapes instead of external image assets, making it
   suitable for GitHub Pages without a build step. */

const canvas = document.querySelector("#board");
const ctx = canvas.getContext("2d");
const W = canvas.width;
const H = canvas.height;

const ROWS = 5;
const COLS = 9;
const TILE_W = 100;
const TILE_H = 92;
const BOARD_X = 98;
const BOARD_Y = 75;
const HOUSE_X = 15;

const plantTypes = {
  seed: {
    name: "Sunbud",
    cost: 50,
    cooldown: 900,
    color: "#e9c657",
    description: "Steady ranged attack",
    damage: 20,
    interval: 1400,
    projectile: "#f6d971"
  },
  wall: {

    damage: 0,
    health: 500
  },
  frost: {
    name: "Frostleaf",
    cost: 100,
    cooldown: 1500,
    color: "#82cad0",
    description: "Slows enemies",

function makeCard(kind) {
  const p = plantTypes[kind];
  const card = document.createElement("button");
  card.className = "card";
  card.dataset.kind = kind;
  card.innerHTML = `
    <span class="card-art"></span>
    <strong>${p.name}</strong>
    <small>${p.cost} sun · ${p.description}</small>
  `;
  card.onclick = () => {
    if ((state.cooldowns[kind] || 0) > 0) {
      setMessage("That ally is still recovering.");
      return;
    }
    state.selected = kind;
    document.querySelectorAll(".card").forEach(c => c.classList.remove("selected"));
    card.classList.add("selected");
    setMessage(`${p.name} selected. Choose a garden tile.`);
  };
  return card;
}

function resetGame() {
  state.running = true;
  state.paused = false;
  state.lastTime = performance.now();
  state.elapsed = 0;
  state.sun = 150;
  state.wave = 1;
  state.score = 0;
  state.spawnTimer = 0;
  state.waveTimer = 0;
  state.plants = [];
  state.enemies = [];
  state.projectiles = [];
  state.particles = [];
  state.cooldowns = {};
  showScreen("#gameScreen");
  setMessage("Plant your first ally.");
  requestAnimationFrame(loop);
}

function tileAt(clientX,clientY) {
  const rect = canvas.getBoundingClientRect();
  const x = (clientX - rect.left) * W / rect.width;
  const y = (clientY - rect.top) * H / rect.height;
  const col = Math.floor((x - BOARD_X) / TILE_W);
  const row = Math.floor((y - BOARD_Y) / TILE_H);
  if (row < 0 || row >= ROWS || col < 0 || col >= COLS) return null;
  return {row,col};
}

canvas.addEventListener("click", event => {
  if (!state.running || state.paused) return;
  const tile = tileAt(event.clientX,event.clientY);
  if (!tile) return;

  const existing = state.plants.find(p => p.row === tile.row && p.col === tile.col);
  const kind = state.selected;
  const type = plantTypes[kind];

  if (existing) {
    if (existing.kind === kind && existing.level < 3) {
      existing.level++;
      existing.maxHealth = Math.round(existing.maxHealth * 1.65);
      existing.health = existing.maxHealth;
      existing.damage = Math.round(existing.damage * 1.6);
      burst(existing.x,existing.y,"#fff2a2",18);
      setMessage(`${type.name} fused to level ${existing.level}.`);
      return;
    }
    setMessage("That tile is occupied.");
    return;
  }

  if (state.sun < type.cost) {
    setMessage("Not enough sun.");
    return;
  }

  state.sun -= type.cost;
  state.cooldowns[kind] = type.cooldown;
  const x = BOARD_X + tile.col * TILE_W + TILE_W / 2;
  const y = BOARD_Y + tile.row * TILE_H + TILE_H / 2;
  const health = type.health || 160;

  state.plants.push({
    kind,
    row:tile.row,
    col:tile.col,
    x,y,
    level:1,
    health,
    maxHealth:health,
    damage:type.damage || 0,
    timer:Math.random() * 500
  });

  burst(x,y,type.color,10);
});

function spawnEnemy() {
  const row = Math.floor(Math.random() * ROWS);
  const elite = state.wave >= 3 && Math.random() < .17;
  const health = elite ? 360 + state.wave * 35 : 150 + state.wave * 24;
  state.enemies.push({
    row,
    x:W + 40,
    y:BOARD_Y + row * TILE_H + TILE_H / 2,
    health,
    maxHealth:health,
    speed:elite ? 12 : 18 + state.wave * 1.5,
    damage:elite ? 38 : 24,
    attackTimer:0,
    slow:0,
    elite
  });
}

function firePlant(p) {
  const type = plantTypes[p.kind];
  const target = state.enemies
    .filter(e => e.row === p.row && e.x > p.x && e.health > 0)
    .sort((a,b) => a.x-b.x)[0];

  if (!target || !type.damage) return;
  state.projectiles.push({
    x:p.x + 26,
    y:p.y - 4,
    row:p.row,
    speed:260,
    damage:p.damage * p.level,
    color:type.projectile,
    frost:p.kind === "frost",
    radius:p.kind === "burst" ? 10 : 7
  });
}

function update(dt) {
  state.elapsed += dt;
  state.waveTimer += dt;

  if (state.waveTimer > state.waveLength) {
    state.waveTimer = 0;
    state.wave++;
    setMessage(`Wave ${state.wave}: the garden grows more dangerous.`);
  }

  state.sun += dt * .006;

  for (const kind in state.cooldowns) {
    state.cooldowns[kind] = Math.max(0,state.cooldowns[kind] - dt);
  }

  state.spawnTimer -= dt;
  const spawnDelay = Math.max(650,2600 - state.wave * 150);
  if (state.spawnTimer <= 0) {
    spawnEnemy();
    state.spawnTimer = spawnDelay * (.7 + Math.random() * .55);
  }

  for (const p of state.plants) {
    p.timer -= dt;
    if (p.timer <= 0) {
      firePlant(p);
      const base = plantTypes[p.kind].interval || 1000;
      p.timer = base / (1 + (p.level - 1) * .25);
    }
  }

  for (const projectile of state.projectiles) {
    projectile.x += projectile.speed * dt / 1000;
    const target = state.enemies.find(e =>
      e.row === projectile.row &&
      Math.abs(e.x - projectile.x) < 22 &&
      e.health > 0
    );
    if (target) {
      target.health -= projectile.damage;
      if (projectile.frost) target.slow = 1800;
      projectile.dead = true;
      burst(projectile.x,projectile.y,projectile.color,4);
      if (target.health <= 0) {
        state.score += target.elite ? 40 : 15;
        burst(target.x,target.y,target.elite ? "#cf7555":"#859e53",12);
      }
    }
    if (projectile.x > W + 30) projectile.dead = true;
  }
  state.projectiles = state.projectiles.filter(p => !p.dead);

  for (const enemy of state.enemies) {
    enemy.slow = Math.max(0,enemy.slow - dt);
    const speedFactor = enemy.slow > 0 ? .42 : 1;
    const tileCol = Math.floor((enemy.x - BOARD_X) / TILE_W);
    const blocker = state.plants.find(p =>
      p.row === enemy.row &&
      p.col === tileCol &&
      Math.abs(enemy.x - p.x) < 42
    );

    if (blocker) {
      enemy.attackTimer -= dt;
      if (enemy.attackTimer <= 0) {
        blocker.health -= enemy.damage;
        enemy.attackTimer = 850;
        burst(blocker.x,blocker.y,"#c8664a",3);
        if (blocker.health <= 0) {
          blocker.dead = true;
          burst(blocker.x,blocker.y,"#76563e",12);
        }
      }
    } else {
      enemy.x -= enemy.speed * speedFactor * dt / 1000;
    }

    if (enemy.x < HOUSE_X + 28) {
      endGame(false);
      return;
    }
  }

  state.enemies = state.enemies.filter(e => e.health > 0 && e.x > -50);
  state.plants = state.plants.filter(p => !p.dead);

  for (const particle of state.particles) {
    particle.life -= dt;
    particle.x += particle.vx * dt / 1000;
    particle.y += particle.vy * dt / 1000;
    particle.vy += 28 * dt / 1000;
  }
  state.particles = state.particles.filter(p => p.life > 0);

  updateStats();
}

function updateStats() {
  $("#sunValue").textContent = Math.floor(state.sun);
  $("#waveValue").textContent = state.wave;
  $("#scoreValue").textContent = state.score;
}

function burst(x,y,color,count) {
  for (let i=0;i<count;i++) {
    const angle = Math.random() * Math.PI * 2;
    const speed = 25 + Math.random() * 80;
    state.particles.push({
      x,y,color,
      vx:Math.cos(angle)*speed,
      vy:Math.sin(angle)*speed-30,
      size:3+Math.random()*5,
      life:300+Math.random()*500,
      maxLife:800
    });
  }
}

function roundedRect(x,y,w,h,r,fill,stroke="#17251f",line=3) {
  ctx.beginPath();
  ctx.roundRect(x,y,w,h,r);
  ctx.fillStyle=fill;
  ctx.fill();
  ctx.lineWidth=line;
  ctx.strokeStyle=stroke;
  ctx.stroke();
}

  for (let row=0;row<ROWS;row++) {

      ctx.strokeStyle="rgba(23,37,31,.18)";
      ctx.lineWidth=2;
      ctx.strokeRect(x,y,TILE_W,TILE_H);
    }
  }
}

function drawPlant(p) {
  const type=plantTypes[p.kind];
  const scale=1+(p.level-1)*.11;
  ctx.save();
  ctx.translate(p.x,p.y);
  ctx.scale(scale,scale);

  ctx.fillStyle="rgba(23,37,31,.2)";
  ctx.beginPath();
  ctx.ellipse(0,29,31,8,0,0,Math.PI*2);
  ctx.fill();

  ctx.fillStyle="#527d42";
  ctx.beginPath();
  ctx.ellipse(-16,17,17,8,-.5,0,Math.PI*2);
  ctx.ellipse(16,17,17,8,.5,0,Math.PI*2);
  ctx.fill();
  ctx.strokeStyle="#17251f";
  ctx.lineWidth=3;
  ctx.stroke();

  ctx.fillStyle=type.color;
  ctx.beginPath();
  ctx.arc(0,-2,25,0,Math.PI*2);
  ctx.fill();
  ctx.stroke();

  if (p.kind === "wall") {
    ctx.fillStyle="#70483b";
    ctx.fillRect(-13,-15,26,31);
    ctx.strokeRect(-13,-15,26,31);
    ctx.strokeStyle="#e1a374";
    ctx.beginPath();
    ctx.moveTo(-7,-10);ctx.lineTo(7,-1);ctx.lineTo(-4,9);
    ctx.stroke();
  } else {
    ctx.fillStyle="#fff1bd";
    ctx.beginPath();
    ctx.arc(-8,-5,5,0,Math.PI*2);
    ctx.arc(8,-5,5,0,Math.PI*2);
    ctx.fill();
    ctx.stroke();
    ctx.fillStyle="#17251f";
    ctx.beginPath();
    ctx.arc(-8,-5,2,0,Math.PI*2);
    ctx.arc(8,-5,2,0,Math.PI*2);
    ctx.fill();
    ctx.beginPath();
    ctx.arc(0,5,9,0,Math.PI);
    ctx.stroke();
  }

  if (p.level > 1) {
    ctx.fillStyle="#fff0a8";
    ctx.font="bold 15px Georgia";
    ctx.textAlign="center";
    ctx.fillText("F"+p.level,0,-34);
  }
  ctx.restore();

  const hp=Math.max(0,p.health/p.maxHealth);
  ctx.fillStyle="#273b2c";
  ctx.fillRect(p.x-27,p.y+35,54,6);
  ctx.fillStyle="#74bd62";
  ctx.fillRect(p.x-27,p.y+35,54*hp,6);
}

function drawEnemy(e) {
  ctx.save();
  ctx.translate(e.x,e.y);
  const body=e.elite ? "#925347" : "#6d8652";
  ctx.fillStyle="rgba(23,37,31,.2)";
  ctx.beginPath();
  ctx.ellipse(0,29,30,8,0,0,Math.PI*2);
  ctx.fill();

  ctx.fillStyle=body;
  ctx.beginPath();
  ctx.roundRect(-24,-28,48,54,16);
  ctx.fill();
  ctx.strokeStyle="#17251f";
  ctx.lineWidth=4;
  ctx.stroke();

  ctx.fillStyle="#d9e1b7";
  ctx.beginPath();
  ctx.arc(-9,-8,6,0,Math.PI*2);
  ctx.arc(9,-8,6,0,Math.PI*2);
  ctx.fill();
  ctx.stroke();

  ctx.fillStyle="#17251f";
  ctx.beginPath();
  ctx.arc(-9,-8,2,0,Math.PI*2);
  ctx.arc(9,-8,2,0,Math.PI*2);
  ctx.fill();

  ctx.strokeStyle="#17251f";
  ctx.beginPath();
  ctx.moveTo(-10,10);ctx.lineTo(10,10);
  ctx.stroke();

  if (e.slow > 0) {
    ctx.strokeStyle="#c9ffff";
    ctx.lineWidth=4;
    ctx.beginPath();
    ctx.arc(0,0,34,0,Math.PI*2);
    ctx.stroke();
  }

  ctx.fillStyle="#263c2b";
  ctx.fillRect(-28,-40,56,6);
  ctx.fillStyle="#d36350";
  ctx.fillRect(-28,-40,56*Math.max(0,e.health/e.maxHealth),6);
  ctx.restore();
}

function drawProjectile(p) {
  ctx.fillStyle=p.color;
  ctx.beginPath();
  ctx.arc(p.x,p.y,p.radius,0,Math.PI*2);
  ctx.fill();
  ctx.strokeStyle="#17251f";
  ctx.lineWidth=2;
  ctx.stroke();
}

function drawParticles() {
  for (const p of state.particles) {
    ctx.globalAlpha=Math.max(0,p.life/p.maxLife);
    ctx.fillStyle=p.color;
    ctx.fillRect(p.x,p.y,p.size,p.size);
  }
  ctx.globalAlpha=1;
}

function render() {
  drawBackground();
  state.plants.forEach(drawPlant);
  state.enemies.forEach(drawEnemy);
  state.projectiles.forEach(drawProjectile);
  drawParticles();

  if (state.paused) {
    ctx.fillStyle="rgba(23,37,31,.65)";
    ctx.fillRect(0,0,W,H);
    ctx.fillStyle="#fff9e8";
    ctx.font="bold 42px Georgia";
    ctx.textAlign="center";
    ctx.fillText("Garden Paused",W/2,H/2);
  }
}

function loop(now) {
  if (!state.running) return;
  const dt=Math.min(50,now-state.lastTime);
  state.lastTime=now;
  if (!state.paused) update(dt);
  render();
  requestAnimationFrame(loop);
}

function endGame(won) {
  state.running=false;
  openModal(
    won ? "Garden Saved" : "The House Was Overrun",
    won
      ? `Your final score was ${state.score}.`
      : `The garden survived ${state.wave} waves with a score of ${state.score}.`,
    "Return to Menu",
    () => showScreen("#menuScreen")
  );
}
